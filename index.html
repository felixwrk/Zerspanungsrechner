<script>

  const v = id => parseFloat(document.getElementById(id).value) || 0;

  const set = (id, text) => document.getElementById(id).textContent = text;

  const r = (x, n = 1) => (isFinite(x) ? x.toFixed(n) : "–");


  function rechne() {

    // Fräsen

    const fn = (v("fVc") * 1000) / (Math.PI * v("fD"));

    set("fN", "Drehzahl n = " + r(fn, 0) + " 1/min");

    set("fVf", "Vorschub vf = " + r(fn * v("fFz") * v("fZ"), 0) + " mm/min");

    set("fVc2", "Schnittgeschwindigkeit vc = " + r(Math.PI * v("fD") * v("fN2") / 1000, 1) + " m/min");

    document.getElementById("fCircle").setAttribute("r", Math.max(10, Math.min(50, v("fD") * 2.5)));


    // Drehen

    const dn = (v("dVc") * 1000) / (Math.PI * v("dD"));

    set("dN", "Drehzahl n = " + r(dn, 0) + " 1/min");

    set("dVf", "Vorschub vf = " + r(dn * v("dF"), 0) + " mm/min");

    set("dVc2", "Schnittgeschwindigkeit vc = " + r(Math.PI * v("dD") * v("dN2") / 1000, 1) + " m/min");

  }


  document.querySelectorAll("input").forEach(i => i.addEventListener("input", rechne));

  rechne();

</script>
